1000000.feels("like a suspiciously round number")         // true

// a class = several judgments in ONE request → one typed value
class Triage {
  urgent: bool       @description("Needs immediate attention?"),
  route: Team,
  churn_risk: float  @description("Likely to churn over this?"),
  mood: "calm" | "concerned" | "angry",
}
ticket.matches<Triage>()
// Triage { urgent: true, route: Technical, churn_risk: 0.56, mood: "angry" }
