vibes.baml
// no new language. feels is a method on every type:

implements<S> Vibes for S {
  function feels(self, quality: string) -> bool {
    Feels(state_of(self), quality)
  }
}

// the return type is the question. bool → Jev
function Feels(state: string, quality: string) -> bool {
  client: "typesafeai/jev-latest"
  prompt: `${role("instructions")} Does this feel ${quality}?
           ${role("user")} ${state}`
}
